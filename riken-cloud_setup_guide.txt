git clone https://github.com/RIKEN-RCCS/benchpark
cd benchpark
. ./setup-env.sh

#Use python3.12 on fx700 nodes (maybe not necessary for all fx nodes)
# Python setup depends on fx nodes, you have to specify a high enough version of Python 
# [h30002@fx17 ~]$ python3 -V
# Python 3.6.8
# [h30002@fx17 ~]$ which python3
# /usr/bin/python3
# [h30002@fx17 ~]$ ls /usr/bin/python3*
# /usr/bin/python3  /usr/bin/python3.11  /usr/bin/python3.12  /usr/bin/python3.6  /usr/bin/python3.6m
#
# benchpark requires Python >=3.8
# Spack does Python >=3.6
# Ramble does Python >=3.6



python3 -m venv .venv
#------------------------------------------------
# if python verson is not enough.
python3.$(ls /usr/bin/python3.[0-9]* 2>/dev/null | sed -E 's#.*/python3\.##' | sort -nr | head -1) -m venv .venv
#-------------------------------------------------

source .venv/bin/activate
pip install --upgrade pip
pip install -r requirements.txt
pip install .[analyze]

export BM="saxpy"
export SYS="riken-cloud"
export WSDIR="workspace"
export CLUSTER="fx700"
export COMP="fj"

# On gh200 clusters cuda/nvhpc version can be added as argument just like for compiler
# On fx700, compiler options: fj,gcc,clang
# On gh200, compiler options: gcc,nvhpc,cuda

benchpark system init --dest=${SYS}-${CLUSTER}-${COMP} ${SYS} cluster=${CLUSTER} compiler=${COMP}
benchpark experiment init ${SYS}-${CLUSTER}-${COMP} ${BM}+mpi+openmp
benchpark setup ${SYS}-${CLUSTER}-${COMP}/${BM} ${WSDIR}
. ./${WSDIR}/setup.sh


# Somme patches for fujitsu externals and clang compiler (Run once after initial spack/ramble/spack-packages setup
wget https://raw.githubusercontent.com/SBA0486/spack-packages/refs/heads/develop/repos/spack_repo/builtin/packages/hpl/package.py -O ${WSDIR}/spack-packages/repos/spack_repo/builtin/packages/hpl/package.py
wget https://raw.githubusercontent.com/SBA0486/spack-packages/refs/heads/develop/repos/spack_repo/builtin/packages/fujitsu_mpi/package.py -O ${WSDIR}/spack-packages/repos/spack_repo/builtin/packages/fujitsu_mpi/package.py
sed -i -e 's@SYSTEM_PATHS = \[\(.*\)\]@SYSTEM_PATHS = [\1, "/opt/FJSVstclanga/cp-1.0.30.01"]@g'  ${WSDIR}/spack/lib/spack/spack/util/environment.py
sed -i -e 's@%fj"):@%fj") or (spec.target == "a64fx" and spec.satisfies("%clang\@11:")):@g'  ${WSDIR}/spack-packages/repos/spack_repo/builtin/packages/cmake/package.py


# setup
ramble --workspace-dir ./${WSDIR}/${SYS}-${CLUSTER}-${COMP}/${BM}/workspace workspace setup

ramble --workspace-dir ./${WSDIR}/${SYS}-${CLUSTER}-${COMP}/${BM}/workspace on
ramble --workspace-dir ./${WSDIR}/${SYS}-${CLUSTER}-${COMP}/${BM}/workspace workspace analyze
